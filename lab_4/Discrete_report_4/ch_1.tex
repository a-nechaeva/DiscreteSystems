\chapter{Синтез апериодического регулятора.}
\label{ch:chap1}

\section{Цель работы}
Изучение различных дискретных алгоритмов управления с заданными характеристиками переходных процессов.
\section{Исходные данные}
Период дискретизации $T = 0.2$ и параметры объекта управления $a = 2$, $b = 4$, $\zeta = 0.7$, $\omega_d = 5$ и $K_v=0.25$.

\section{Ход работы}
Объект управления задан непрерывной передаточной функцией
\begin{equation}
	G(s) = \frac{e^{-as}}{1+bs} = \frac{e^{-2s}}{1+4s}
\end{equation}
Для непрерывного ОУ синтезируем апериодический регулятор. Период дискретизации $T=1$.

Дискретизованная передаточная функция ОУ с ЭНП:
\begin{multline}
	HG(z) = Z \left\{\frac{1-e^{-sT}}{s} G(s)\right\} = \left(1 - z^{-1}\right) Z \left\{\frac{e^{-2s}}{s(1+4s)}\right\} =\\ =
	\left(1 - z^{-1}\right) z^{-2}  Z \left\{\frac{1}{s(1+4s)}\right\} =
		\left(1 - z^{-1}\right) z^{-2}  Z \left\{\frac{0.25}{s(0.25+s)}\right\} = \\ =
			\left(1 - z^{-1}\right) z^{-2} \frac{z\left(1 - e^{-0.25}\right)}{(z-1) \left(z - e^{-0.25}\right)} =
			\frac{z^{-2} \left(1 - e^{-0.25}\right)}{\left(z - e^{-0.25}\right)} = \\ =
			\frac{z^{-3} \left(1 - e^{-0.25}\right)}{1 - e^{-0.25} z^{-1}} = \frac{1 - e^{-0.25}}{z^3 - e^{-0.25} z^{2}} \approx
			\frac{0.2212}{z^3 - 0.7788z^2}
\end{multline}
Получим передаточную функцию регулятора
\begin{multline}
	D(z) = \frac{1}{HG(z)} \frac{T(z)}{1-T(z)} = \frac{1}{HG(z)} \frac{z^{-k}}{1 - z^{-k}} = \\ =
	\frac{1 - e^{-0.25} z^{-1}}{z^{-3} \left(1 - e^{-0.25}\right)} \frac{z^{-3}}{1 - z^{-3}} =
	\frac{z^3 - e^{-0.25} z^{2}}{ \left(1 - e^{-0.25}\right) \left(z^3 - 1\right)} \approx 
	\frac{z^3 - 0.7788 z^{2}}{ 0.2212 \left(z^3 - 1\right)}
\end{multline}
