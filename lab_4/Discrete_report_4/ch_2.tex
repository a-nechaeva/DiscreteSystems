\chapter{Синтез регулятора Далина}
\label{ch:chap2}
Для непрерывного ОУ синтезируем регулятор Далина. Период дискретизации $T=1$.

Регулятор Далина представляет собой модификацию апериодического регулятора и обеспечивает более плавный экспоненциальный отклик, чем у последнего. Желаемое поведение системы в $s$-плоскости задается выражением
\begin{equation}
	Y(s) = \frac{1}{s} \frac{e^{as}}{1 + qs},
\end{equation}
где параметры $a$ и $q$ определяют численные параметры желаемого поведения выходной величины.

Осуществим $Z$-преобразование желаемой реакции системы, приняв $a=kT$:
\begin{equation}
	Y(z) = \frac{z^{-k-1} \left(1 - e^{-\frac{T}{q}}\right)}{\left(1 - z^{-1}\right) \left(1-e^{-\frac{T}{q}}z^{-1}\right)}
\end{equation}
Тогда желаемая передаточная функция замкнутой системы будет иметь вид
\begin{equation}
	T(z) = \frac{Y(z)}{R(z)} = \frac{z^{-k-1} \left(1 - e^{-\frac{T}{q}}\right) \left(1-z^{-1}\right)}{\left(1-z^{-1}\right) \left(1-e^{-\frac{T}{q}}z^{-1}\right)} = \frac{z^{-k-1} \left(1-e^{-\frac{T}{q}}\right)}{1 - e^{-\frac{T}{q}}z^{-1}}
\end{equation}
Передаточная функция требуемого регулятора будет иметь вид
\begin{equation}
	D(z) = \frac{1}{HG(z)} \frac{z^{-k-1} \left(1-e^{-\frac{T}{q}}\right)}{1-e^{-\frac{T}{q}}z^{-1} -\left(1-e^{-\frac{T}{q}}\right)z^{-k-1}}
\end{equation}
Дискретную передаточную функцию ОУ с ЭНП перепишем с прошлой части работы
\begin{equation}
	HG(z) =
	\frac{z^{-3} \left(1 - e^{-0.25}\right)}{1 - e^{-0.25} z^{-1}} = \frac{1 - e^{-0.25}}{z^3 - e^{-0.25} z^{2}} \approx
	\frac{0.2212}{z^3 - 0.7788z^2}
\end{equation}
Передаточная функции регулятора при $q = 4$
\begin{equation}
	D(z) = \frac{1 - e^{-0.25} z^{-1}}{z^{-3} \left(1 - e^{-0.25}\right)} \frac{z^{-k-1} \left(1-e^{-0.25}\right)}{1-e^{-0.25}z^{-1} -\left(1-e^{-0.25}\right)z^{-k-1}}
\end{equation}
С учетом требования физической реализуемости выберем $k = 2$ и получим
\begin{multline}
	D(z) = \frac{1 - e^{-0.25} z^{-1}}{z^{-3} \left(1 - e^{-0.25}\right)} \frac{z^{-3} \left(1-e^{-0.25}\right)}{1-e^{-0.25}z^{-1} -\left(1-e^{-0.25}\right)z^{-3}} =\\
	= \frac{1 - e^{-0.25} z^{-1}}{1-e^{-0.25}z^{-1} -\left(1-e^{-0.25}\right)z^{-3}} =
	\frac{z^3 - e^{-0.25} z^{2}}{z^3-e^{-0.25}z^{2} -\left(1-e^{-0.25}\right)}  \approx \\ \approx
		\frac{z^3 - 0.7788 z^{2}}{z^3-0.7788z^{2} -0.2212}
\end{multline}
